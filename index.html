service cloud.firestore {
  match /databases/{db}/documents {
    function signedIn() { return request.auth != null; }
    function userDoc() { return get(/databases/$(db)/documents/users/$(request.auth.uid)); }
    function hasUserDoc() { return exists(/databases/$(db)/documents/users/$(request.auth.uid)); }
    function isAdmin() {
      return signedIn() && (request.auth.token.email in ['teamhr@ggl.com']
        || (hasUserDoc() && userDoc().data.role == 'admin'));
    }
    function isActive() { return signedIn() && (!hasUserDoc() || userDoc().data.active != false); }
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow write: if isAdmin();
    }
    match /app/salaries { allow read, write: if isAdmin(); }
    match /app/salhist  { allow read, write: if isAdmin(); }
    match /{document=**} { allow read, write: if isActive(); }
  }
}
